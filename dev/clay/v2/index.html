Cómo integrarlo en tu sistema

<!-- 1. Incluye el CSS -->
<link rel="stylesheet" href="clay-techrevx.css">

<!-- 2. Set mode + color on <html> -->
<html class="tx-light">

<!-- 3. Usa las clases .clay-* -->
<button class="clay-btn clay-btn--primary">Sign in</button>
<input class="clay-input" placeholder="Username">
<div class="clay-card">...</div>

<script src="clay-techrevx.js" defer></script>
<script>
// 4. Persistencia (compat con tu app APEX)
clayTheme.apply("light", "teal");
</script>
